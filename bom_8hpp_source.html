132 inline const bom_info*
133 find(type bom_type)
134 {
135 const auto& bom_array = table();
136
137 for (std::size_t i = 0; i < bom_array_size; ++i)
138 {
139 if (bom_array[i].type == bom_type)
140 {
141 return &bom_array[i];
142 }
143 }
144
145 return nullptr;
146 }
147 }

157 inline std::optional<type>
158 detect(const char* input, std::size_t length)
159 {
160 const auto& bom_array = internal::table();
161
162 for (std::size_t i = 0; i < internal::bom_array_size; ++i)
163 {
164 const auto& info = bom_array[i];
165
166 if (length < info.length)
167 {
168 continue;
169 }
170 else if (!std::memcmp(input, info.bytes, info.length))
171 {
172 return info.type;
173 }
174 }
175
176 return std::nullopt;
177 }

195 inline std::string_view
