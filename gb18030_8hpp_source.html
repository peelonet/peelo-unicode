38 // GB 18030 byte ranges:
39 // 1-byte: 0x00-0x7F (ASCII)
40 // 2-byte: [0x81-0xFE][0x40-0x7E | 0x80-0xFE]
41 // 4-byte: [0x81-0xFE][0x30-0x39][0x81-0xFE][0x30-0x39]
42
43 // Binary search in two-byte table by GB code (for decoding)
44 inline char32_t
45 lookup_two_byte_to_unicode(std::uint16_t gb_code)
46 {
47 std::size_t low = 0;
48 std::size_t high = tables::gb_to_unicode.size();
49
50 while (low < high)
51 {
52 const auto mid = low + (high - low) / 2;
53 if (tables::gb_to_unicode[mid].gb < gb_code)
54 {
55 low = mid + 1;
56 }
57 else if (tables::gb_to_unicode[mid].gb > gb_code)
58 {
59 high = mid;
60 }
61 else
62 {
63 return tables::gb_to_unicode[mid].unicode;
64 }
65 }
66
67 return 0xFFFFFFFF; // Not found
68 }

70 // Binary search by Unicode codepoint (for encoding)
71 inline std::uint16_t
72 lookup_unicode_to_two_byte(char32_t unicode)
73 {
74 std::size_t low = 0;
75 std::size_t high = tables::unicode_to_gb.size();
76
77 while (low < high)
78 {
79 const auto mid = low + (high - low) / 2;
80 if (tables::unicode_to_gb[mid].unicode < unicode)
81 {
82 low = mid + 1;
83 }
84 else if (tables::unicode_to_gb[mid].unicode > unicode)
85 {
86 high = mid;
87 }
88 else
89 {
90 return tables::unicode_to_gb[mid].gb;
91 }
92 }
93
94 return 0; // Not found
95 }

97 // Convert 4-byte GB18030 sequence to linear offset
98 inline std::uint32_t
99 four_byte_to_offset(
100 unsigned char b1,
101 unsigned char b2,
102 unsigned char b3,
103 unsigned char b4
104 )
105 {
106 return ((b1 - 0x81) * 12600) +
107 ((b2 - 0x30) * 1260) +
108 ((b3 - 0x81) * 10) +
109 (b4 - 0x30);
110 }

112 // Convert linear offset to 4-byte GB18030 sequence
113 inline void
114 offset_to_four_byte(
115 std::uint32_t offset,
116 unsigned char& b1,
117 unsigned char& b2,
118 unsigned char& b3,
119 unsigned char& b4
120 )
121 {
122 b4 = static_cast<unsigned char>((offset % 10) + 0x30);
123 offset /= 10;
124 b3 = static_cast<unsigned char>((offset % 126) + 0x81);
125 offset /= 126;
126 b2 = static_cast<unsigned char>((offset % 10) + 0x30);
127 offset /= 10;
128 b1 = static_cast<unsigned char>(offset + 0x81);
129 }

131 // Convert 4-byte GB offset to Unicode using ranges table
132 inline char32_t
133 four_byte_offset_to_unicode(std::uint32_t offset)
134 {
135 // Find the range that contains this offset
136 // The ranges are sorted by gb_offset, so we can find the last range
137 // where gb_offset <= our offset
138 std::size_t range_idx = 0;
139 for (std::size_t i = 1; i < tables::four_byte_ranges.size(); ++i)
140 {
141 if (tables::four_byte_ranges[i].gb_offset <= offset)
142 {
143 range_idx = i;
144 }
145 else
146 {
147 break;
148 }
149 }
150
151 const auto& range = tables::four_byte_ranges[range_idx];
152 return range.unicode_start + (offset - range.gb_offset);
153 }

155 // Convert Unicode to 4-byte GB offset using ranges table
156 inline std::uint32_t
157 unicode_to_four_byte_offset(char32_t unicode)
158 {
159 // Find the range that contains this unicode codepoint
160 for (std::size_t i = tables::four_byte_ranges.size(); i > 0; --i)
161 {
162 const auto& range = tables::four_byte_ranges[i - 1];
163 if (unicode >= range.unicode_start)
164 {
165 return range.gb_offset + (unicode - range.unicode_start);
166 }
167 }
168 return 0xFFFFFFFF; // Not found
169 }

176 inline bool
177 is_first_byte(unsigned char byte)
178 {
179 return byte <= 0x7F || (byte >= 0x81 && byte <= 0xFE);

200 inline void
201 encode_codepoint(char32_t c, std::string& output)
202 {
203 // ASCII range - 1 byte
204 if (c <= 0x7F)
205 {
206 output.append(1, static_cast<char>(c));
207 return;
208 }
209
210 // Try two-byte encoding first
211 const auto two_byte = internal::lookup_unicode_to_two_byte(c);
212
213 if (two_byte != 0)
214 {
215 output.append(1, static_cast<char>(two_byte >> 8));
216 output.append(1, static_cast<char>(two_byte & 0xFF));
217 return;
218 }
219
220 // Four-byte encoding
221 const auto offset = internal::unicode_to_four_byte_offset(c);
222 if (offset != 0xFFFFFFFF)
223 {
224 unsigned char b1, b2, b3, b4;
225
226 internal::offset_to_four_byte(offset, b1, b2, b3, b4);
227 output.append(1, static_cast<char>(b1));
228 output.append(1, static_cast<char>(b2));
229 output.append(1, static_cast<char>(b3));
230 output.append(1, static_cast<char>(b4));
231 }
232 }

234 inline bool
235 decode_advance(
236 const char* input,
237 std::size_t& i,
238 const std::size_t length,
239 char32_t& result
240 )
241 {
242 if (i >= length)
243 {
244 return false;
245 }
246
247 const auto b1 = static_cast<unsigned char>(input[i]);
248
249 // ASCII - 1 byte
250 if (b1 <= 0x7F)
251 {
252 result = static_cast<char32_t>(b1);
253 ++i;
254 return true;
255 }
256
257 // Must be 0x81-0xFE for multi-byte
258 if (b1 < 0x81 || b1 > 0xFE)
259 {
260 ++i;
261 return false;
262 }
263
264 if (i + 1 >= length)
265 {
266 ++i;
267 return false;
268 }
269
270 const auto b2 = static_cast<unsigned char>(input[i + 1]);
271
272 // Check for 4-byte sequence (second byte is 0x30-0x39)
273 if (is_four_byte_second_fourth(b2))
274 {
275 if (i + 3 >= length)
276 {
277 ++i;
278 return false;
279 }
280
281 const auto b3 = static_cast<unsigned char>(input[i + 2]);
282 const auto b4 = static_cast<unsigned char>(input[i + 3]);
283
284 if (b3 < 0x81 || b3 > 0xFE || !is_four_byte_second_fourth(b4))
285 {
286 ++i;
287 return false;
288 }
289
290 const auto offset = internal::four_byte_to_offset(b1, b2, b3, b4);
291 result = internal::four_byte_offset_to_unicode(offset);
292
293 if (result == 0xFFFFFFFF)
294 {
295 ++i;
296 return false;
297 }
298
299 i += 4;
300 return true;
301 }
302
303 // Two-byte sequence
304 if (!is_two_byte_second(b2))
305 {
306 ++i;
307 return false;
308 }
309
310 const std::uint16_t gb_code = (static_cast<std::uint16_t>(b1) << 8) | b2;
311 result = internal::lookup_two_byte_to_unicode(gb_code);
312
313 if (result == 0xFFFFFFFF)
314 {
315 ++i;
316 return false;
317 }
318
319 i += 2;
320 return true;
321 }
