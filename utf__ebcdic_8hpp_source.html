116 inline std::size_t
117 i8_sequence_length(unsigned char lead)
118 {
119 if (lead <= 0x9f)
120 {
121 return 1;
122 }
123 else if (is_i8_trailing(lead))
124 {
125 return 0;
126 }
127 else if ((lead & 0xe0) == 0xc0)
128 {
129 return 2;
130 }
131 else if ((lead & 0xf0) == 0xe0)
132 {
133 return 3;
134 }
135 else if ((lead & 0xf8) == 0xf0)
136 {
137 return 4;
138 }
139 else if ((lead & 0xfc) == 0xf8)
140 {
141 return 5;
142 }
143
144 // 6- and 7-byte forms exist for UCS-4 beyond Unicode; reject them.
145 return 0;
146 }

299 inline bool
300 decode_advance(
301 const char* input,
302 std::size_t& i,
303 const std::size_t length,
304 char32_t& result
305 )
306 {
307 if (i >= length)
308 {
309 return false;
310 }
311
312 const auto lead_i8 = internal::ebcdic_to_i8[
313 static_cast<unsigned char>(input[i])
314 ];
315 const auto seq_length = internal::i8_sequence_length(lead_i8);
316
317 if (!seq_length || i + seq_length > length)
318 {
319 ++i;
320 return false;
321 }
322
323 if (seq_length == 1)
324 {
325 result = static_cast<char32_t>(lead_i8);
326 ++i;
327 return true;
328 }
329
330 unsigned char i8_bytes[5];
331
332 i8_bytes[0] = lead_i8;
333 for (std::size_t j = 1; j < seq_length; ++j)
334 {
335 i8_bytes[j] = internal::ebcdic_to_i8[
336 static_cast<unsigned char>(input[i + j])
337 ];
338 if (!internal::is_i8_trailing(i8_bytes[j]))
339 {
340 ++i;
341 return false;
342 }
343 }
344
345 switch (seq_length)
346 {
347 case 2:
348 result = static_cast<char32_t>(
349 ((i8_bytes[0] & 0x1f) << 5) | (i8_bytes[1] & 0x1f)
350 );
351 break;
352
353 case 3:
354 result = static_cast<char32_t>(
355 ((i8_bytes[0] & 0x0f) << 10)
356 | ((i8_bytes[1] & 0x1f) << 5)
357 | (i8_bytes[2] & 0x1f)
358 );
359 break;
360
361 case 4:
362 result = static_cast<char32_t>(
363 ((i8_bytes[0] & 0x07) << 15)
364 | ((i8_bytes[1] & 0x1f) << 10)
365 | ((i8_bytes[2] & 0x1f) << 5)
366 | (i8_bytes[3] & 0x1f)
367 );
368 break;
369
370 case 5:
371 result = static_cast<char32_t>(
372 ((i8_bytes[0] & 0x03) << 20)
373 | ((i8_bytes[1] & 0x1f) << 15)
374 | ((i8_bytes[2] & 0x1f) << 10)
375 | ((i8_bytes[3] & 0x1f) << 5)
376 | (i8_bytes[4] & 0x1f)
377 );
378 break;
379
380 default:
381 ++i;
382 return false;
383 }
384
385 i += seq_length;
386
387 return true;
388 }
